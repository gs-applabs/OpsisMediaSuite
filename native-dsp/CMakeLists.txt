cmake_minimum_required(VERSION 3.20)
project(OpsisAudioDSP LANGUAGES CXX)
add_library(opsis_dsp STATIC opsis_dsp.cpp)
target_compile_features(opsis_dsp PUBLIC cxx_std_17)
target_include_directories(opsis_dsp PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(opsis_dsp PRIVATE /W4 /WX /fp:precise)
else()
  target_compile_options(opsis_dsp PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_executable(opsis_dsp_test dsp_test.cpp)
target_link_libraries(opsis_dsp_test PRIVATE opsis_dsp)
add_test(NAME opsis_dsp_test COMMAND opsis_dsp_test)

if(WIN32)
  add_executable(opsis_wasapi_test wasapi_test.cpp)
  target_link_libraries(opsis_wasapi_test PRIVATE opsis_dsp ole32 uuid)
  add_test(NAME opsis_wasapi_test COMMAND opsis_wasapi_test)
  set_tests_properties(opsis_wasapi_test PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 20)
endif()

add_executable(opsis_dsp_pcm pcm_adapter.cpp)
target_link_libraries(opsis_dsp_pcm PRIVATE opsis_dsp)
